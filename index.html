# Send live campus telemetry from Raspberry Pi or BMS
import urllib.request, json, time

API_URL = "http://localhost:5000/api/campus/ingest"

payload = {
    "buildingId": "b1",           # Engineering Complex
    "powerDrawKW": 31.0,          # From Modbus RS485 / CT clamp
    "waterFlowLPM": 6.5,          # From pulse flow meter
    "indoorTempC": 25.5,          # From DHT22 sensor
    "emptyRoomsWithPowerOn": 2    # From PIR motion sensors
}

req = urllib.request.Request(API_URL, data=json.dumps(payload).encode(), headers={"Content-Type": "application/json"})
res = urllib.request.urlopen(req)
print("Live Telemetry Synced:", res.read().decode())
// ESP32 / Arduino Wi-Fi Telemetry Streamer
#include <WiFi.h>
#include <HTTPClient.h>

const char* serverUrl = "http://localhost:5000/api/campus/ingest";

void pushSensorData(float kw, float lpm, int emptyRooms) {
  HTTPClient http;
  http.begin(serverUrl);
  http.addHeader("Content-Type", "application/json");

  String json = "{\"buildingId\":\"b1\",\"powerDrawKW\":" + String(kw) + 
                ",\"waterFlowLPM\":" + String(lpm) + 
                ",\"emptyRoomsWithPowerOn\":" + String(emptyRooms) + "}";

  int code = http.POST(json);
  Serial.println("Response: " + String(code));
  http.end();
}
